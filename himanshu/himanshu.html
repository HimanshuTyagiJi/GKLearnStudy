<script>
const cv = document.getElementById("cv");
const ctx = cv.getContext("2d");

let elements = [];
let selected = null;
let drag = false;
let offsetX, offsetY;

/* ------------ ADD ELEMENTS -------------- */

function addRect(){
    elements.push({type:"rect", x:50, y:50, w:120, h:80, color:"#4caf50"});
    drawAll();
}

function addCircle(){
    elements.push({type:"circle", x:150, y:150, r:45, color:"#ff5722"});
    drawAll();
}

function addText(){
    let t = prompt("Enter text:");
    if(!t) return;
    elements.push({type:"text", x:80, y:80, text:t, size:28, color:"#000"});
    drawAll();
}

function loadImage(e){
    let img = new Image();
    img.onload = () => {
        elements.push({type:"img", x:100, y:100, w:200, h:150, img});
        drawAll();
    };
    img.src = URL.createObjectURL(e.target.files[0]);
}

/* ------------ DRAWING SYSTEM -------------- */

function drawAll(){
    ctx.clearRect(0,0,cv.width,cv.height);
    
    elements.forEach(el => {
        ctx.save();
        ctx.fillStyle = el.color || "#000";
        ctx.strokeStyle = "black";

        if(el.type==="rect"){
            ctx.fillRect(el.x, el.y, el.w, el.h);
        }
        else if(el.type==="circle"){
            ctx.beginPath();
            ctx.arc(el.x, el.y, el.r, 0, Math.PI*2);
            ctx.fill();
        }
        else if(el.type==="text"){
            ctx.font = el.size+"px Arial";
            ctx.fillText(el.text, el.x, el.y);
        }
        else if(el.type==="img"){
            ctx.drawImage(el.img, el.x, el.y, el.w, el.h);
        }

        ctx.restore();
    });
}

/* ------------ DRAGGING -------------- */

cv.addEventListener("mousedown", e => {
    const mx = e.offsetX;
    const my = e.offsetY;

    selected = null;

    for(let i=elements.length-1;i>=0;i--){
        let el = elements[i];

        if(el.type==="rect" &&
           mx>el.x && mx<el.x+el.w &&
           my>el.y && my<el.y+el.h){
            selected = el;
            break;
        }

        if(el.type==="circle" &&
            Math.hypot(mx-el.x, my-el.y) < el.r){
            selected = el;
            break;
        }

        if(el.type==="text" &&
            mx>el.x-10 && mx<el.x+100 &&
            my>el.y-30 && my<el.y+10){
            selected = el;
            break;
        }

        if(el.type==="img" &&
           mx>el.x && mx<el.x+el.w &&
           my>el.y && my<el.y+el.h){
            selected = el;
            break;
        }
    }

    if(selected){
        drag = true;
        offsetX = mx - selected.x;
        offsetY = my - selected.y;
    }
});

cv.addEventListener("mousemove", e => {
    if(!drag || !selected) return;

    selected.x = e.offsetX - offsetX;
    selected.y = e.offsetY - offsetY;

    drawAll();
});

cv.addEventListener("mouseup", () => drag = false);


/* ------------ EXPORT CODE (ctx JS) -------------- */

function exportCode(){
    let code = `const c = document.getElementById("canvasID");\nconst x = c.getContext("2d");\n\n`;

    elements.forEach(el=>{
        if(el.type==="rect"){
            code += `x.fillStyle="${el.color}"; x.fillRect(${el.x},${el.y},${el.w},${el.h});\n`;
        }
        else if(el.type==="circle"){
            code += `x.beginPath(); x.arc(${el.x},${el.y},${el.r},0,Math.PI*2); x.fillStyle="${el.color}"; x.fill();\n`;
        }
        else if(el.type==="text"){
            code += `x.font="${el.size}px Arial"; x.fillStyle="${el.color}"; x.fillText("${el.text}",${el.x},${el.y});\n`;
        }
        else if(el.type==="img"){
            code += `// Image not auto-included, use:\n// let img = new Image(); img.src="your-image.png"; img.onload=()=>{\n// x.drawImage(img,${el.x},${el.y},${el.w},${el.h});\n// }\n\n`;
        }
    });

    document.getElementById("output").value = code;
}

/* ------------ EXPORT PNG -------------- */
function exportPNG(){
    let url = cv.toDataURL("image/png");
    let a = document.createElement("a");
    a.href = url;
    a.download = "canvas.png";
    a.click();
}

/* ------------ ALT TEXT GENERATOR -------------- */
function generateAlt(){
    let alt = "A canvas image containing: ";
    elements.forEach(el => {
        if(el.type==="rect") alt+="a colored rectangle, ";
        if(el.type==="circle") alt+="a circle, ";
        if(el.type==="text") alt+=`text saying '${el.text}', `;
        if(el.type==="img") alt+="an embedded image, ";
    });

    document.getElementById("output").value = alt;
}
