// =================== HELPER FUNCTIONS =======================
function roundRectFill(ctx, x, y, w, h, r, fillStyle){
    ctx.beginPath();
    ctx.moveTo(x+r, y);
    ctx.arcTo(x+w, y, x+w, y+h, r);
    ctx.arcTo(x+w, y+h, x, y+h, r);
    ctx.arcTo(x, y+h, x, y, r);
    ctx.arcTo(x, y, x+w, y, r);
    ctx.closePath();
    ctx.fillStyle = fillStyle;
    ctx.fill();
}

function roundRectStroke(ctx, x, y, w, h, r, strokeStyle, lineWidth){
    ctx.beginPath();
    ctx.moveTo(x+r, y);
    ctx.arcTo(x+w, y, x+w, y+h, r);
    ctx.arcTo(x+w, y+h, x, y+h, r);
    ctx.arcTo(x, y+h, x, y, r);
    ctx.arcTo(x, y, x+w, y, r);
    ctx.closePath();
    ctx.strokeStyle = strokeStyle;
    ctx.lineWidth = lineWidth;
    ctx.stroke();
}

function wrapText(ctx, text, x, y, maxWidth, lineHeight){
    const words = text.split(" ");
    let line = "";
    for(let word of words){
        let test = line + word + " ";
        if(ctx.measureText(test).width > maxWidth){
            ctx.fillText(line, x, y);
            line = word + " ";
            y += lineHeight;
        } else line = test;
    }
    ctx.fillText(line, x, y);
}

// ===================== MAIN DRAW FUNCTION =====================
function drawImg(){
    const cvs = document.getElementById("ctx");
    const cx = cvs.getContext("2d");

    const W = 1280, H = 720;
    cvs.width = W;
    cvs.height = H;

    const text = document.getElementById("img-txt").value;
    const size = parseInt(document.getElementById("img-size").value);
    const c1 = document.getElementById("c1").value;
    const c2 = document.getElementById("c2").value;
    const tc = document.getElementById("tc").value;

    // BG gradient
    let g = cx.createLinearGradient(0,0,W,H);
    g.addColorStop(0,c1);
    g.addColorStop(1,c2);
    cx.fillStyle = g;
    cx.fillRect(0,0,W,H);

    // glow
    let glow = cx.createRadialGradient(W*0.7,H*0.35,10,W*0.7,H*0.35,600);
    glow.addColorStop(0,'rgba(255,255,255,0.12)');
    glow.addColorStop(1,'transparent');
    cx.fillStyle = glow;
    cx.fillRect(0,0,W,H);

    // diagonal lines
    cx.strokeStyle = "rgba(255,255,255,0.06)";
    for(let i=-400;i<1500;i+=45){
        cx.beginPath();
        cx.moveTo(i,0);
        cx.lineTo(i+500,H);
        cx.stroke();
    }

    // glass card
    const cardX = W*0.40, cardY = H*0.2, cardW = W*0.55, cardH = H*0.6;
    roundRectFill(cx,cardX,cardY,cardW,cardH,20,"rgba(255,255,255,0.10)");
    roundRectStroke(cx,cardX,cardY,cardW,cardH,20,"rgba(255,255,255,0.15)",2);

    // draw triangle
    const Ax=200, Ay=550, sizeT=300;
    const Bx=Ax+sizeT, By=Ay;
    const Cx=Ax, Cy=Ay-sizeT*0.6;

    cx.beginPath();
    cx.moveTo(Ax,Ay);
    cx.lineTo(Bx,By);
    cx.lineTo(Cx,Cy);
    cx.closePath();
    cx.fillStyle="rgba(255,255,255,0.12)";
    cx.fill();
    cx.strokeStyle="white";
    cx.lineWidth=3;
    cx.stroke();

    // title
    cx.fillStyle=tc;
    cx.font = `bold ${size}px Poppins`;
    cx.shadowColor="rgba(0,0,0,0.4)";
    cx.shadowBlur=18;
    wrapText(cx,text,cardX+40,cardY+100,cardW-80,size+10);
    cx.shadowBlur=0;

    // branding
    cx.font="22px Poppins";
    cx.fillStyle="rgba(255,255,255,0.9)";
    cx.textAlign="right";
    cx.fillText("GK Learn Study",W-20,H-20);
}

    link.click();
}
